```
lets(
  pn, prop("Points needed"), ph, prop("Team points"),
  mg, prop("Money goal"), mr, if(empty(prop("Raised")), 0, prop("Raised")),
  pOK, empty(pn) or ph >= pn,
  mOK, empty(mg) or mr >= mg,
  pBar, if(empty(pn), "", "⭐ " + repeat("▓", min(10, floor(ph / pn * 10))) + repeat("░", 10 - min(10, floor(ph / pn * 10))) + " " + format(ph) + "/" + format(pn) + " points"),
  mBar, if(empty(mg), "", "💰 " + repeat("▓", min(10, floor(mr / mg * 10))) + repeat("░", 10 - min(10, floor(mr / mg * 10))) + " $" + format(mr) + "/$" + format(mg)),
  if(empty(pn) and empty(mg), "",
    if(pOK and mOK, "🔓 Unlocked · " + join([pBar, mBar].filter(current != ""), " · "),
      join([pBar, mBar].filter(current != ""), "\n")))
)
```
